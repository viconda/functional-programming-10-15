from dataclasses import replace

from .model import Cart, Event, Item


def add_item(cart: Cart, event: Event) -> Cart:
    """Сохранить снимок и добавить товар в новую версию."""
    if not event.sku:
        raise ValueError("empty sku")
    if event.price < 0:
        raise ValueError("negative price")
    if event.qty <= 0:
        raise ValueError("invalid quantity")
    existing = next((x for x in cart.items if x.sku == event.sku), None)
    if existing is not None and existing.price != event.price:
        raise ValueError("price mismatch")
    item = Item(event.sku, event.price, event.qty + (existing.qty if existing else 0))
    return replace(
        cart, items=tuple(sorted((*[x for x in cart.items if x.sku != event.sku], item)))
    )


def remove_item(cart: Cart, event: Event) -> Cart:
    """Удалить количество с проверкой достаточности."""
    if event.qty <= 0:
        raise ValueError("invalid quantity")
    old = next((x for x in cart.items if x.sku == event.sku), None)
    if old is None:
        raise ValueError("unknown sku")
    if event.qty > old.qty:
        raise ValueError("insufficient quantity")
    kept = tuple(x for x in cart.items if x.sku != event.sku)
    items = kept + ((replace(old, qty=old.qty - event.qty),) if event.qty < old.qty else ())
    return replace(cart, items=tuple(sorted(items)))


def reduce_event(cart: Cart, event: Event) -> Cart:
    """Конечный автомат корзины без побочных эффектов."""
    if cart.closed:
        raise ValueError("cart closed")
    if event.op == "add":
        return add_item(cart, event)
    if event.op == "remove":
        return remove_item(cart, event)
    if event.op == "promo":
        if event.code not in ("", "SAVE10", "FIX500"):
            raise ValueError("unknown promo")
        return replace(cart, promo=event.code)
    if event.op == "checkout":
        if not cart.items:
            raise ValueError("empty cart")
        return replace(cart, closed=True)
    raise ValueError("unknown operation")


src\shop\discounts.py
from collections.abc import Callable

from .model import Cart


def subtotal(cart: Cart) -> int:
    """Сумма строк корзины."""
    return sum(x.price * x.qty for x in cart.items)


def volume_rule(cart: Cart) -> int:
    """Пять процентов от 10000 тиынов."""
    amount = subtotal(cart)
    return amount * 5 // 100 if amount >= 10000 else 0


def promo_rule(cart: Cart) -> int:
    """Скидка промокода с целочисленным округлением вниз."""
    amount = subtotal(cart)
    return (
        amount * 10 // 100
        if cart.promo == "SAVE10"
        else min(500, amount)
        if cart.promo == "FIX500"
        else 0
    )


def discount(
    cart: Cart, rules: tuple[Callable[[Cart], int], ...] = (volume_rule, promo_rule)
) -> int:
    """Композиция правил, выбирающая наибольшую скидку."""
    return max((rule(cart) for rule in rules), default=0)


src\shop\legacy.py
"""Исходная императивная версия магазина, разработанная для исследования."""

import json
from pathlib import Path
from typing import Any


def new_cart() -> dict[str, Any]:
    """Создать изменяемую корзину."""
    cart: dict[str, Any] = {}
    cart["items"] = []
    cart["promo"] = ""
    cart["closed"] = False
    return cart


def check_open(cart: dict[str, Any]) -> None:
    """Отклонить изменение оформленной корзины."""
    if cart["closed"]:
        raise ValueError("cart closed")


def add_item(cart: dict[str, Any], event: dict[str, Any]) -> None:
    """Добавить товар, изменив список строк корзины."""
    check_open(cart)
    sku = str(event.get("sku", "")).strip()
    price = int(event.get("price", 0))
    quantity = int(event.get("qty", 1))
    if not sku:
        raise ValueError("empty sku")
    if price < 0:
        raise ValueError("negative price")
    if quantity <= 0:
        raise ValueError("invalid quantity")
    for item in cart["items"]:
        if item["sku"] == sku:
            if item["price"] != price:
                raise ValueError("price mismatch")
            item["qty"] += quantity
            return
    row: dict[str, Any] = {}
    row["sku"] = sku
    row["price"] = price
    row["qty"] = quantity
    cart["items"].append(row)


def remove_item(cart: dict[str, Any], event: dict[str, Any]) -> None:
    """Удалить количество или всю строку товара."""
    check_open(cart)
    sku = str(event.get("sku", "")).strip()
    quantity = int(event.get("qty", 1))
    if quantity <= 0:
        raise ValueError("invalid quantity")
    for index in range(len(cart["items"])):
        row = cart["items"][index]
        if row["sku"] == sku:
            if quantity > row["qty"]:
                raise ValueError("insufficient quantity")
            row["qty"] -= quantity
            if row["qty"] == 0:
                cart["items"].pop(index)
            return
    raise ValueError("unknown sku")


def set_promo(cart: dict[str, Any], event: dict[str, Any]) -> None:
    """Установить один промокод."""
    check_open(cart)
    code = str(event.get("code", "")).upper().strip()
    if code not in ("", "SAVE10", "FIX500"):
        raise ValueError("unknown promo")
    cart["promo"] = code


def subtotal(cart: dict[str, Any]) -> int:
    """Подсчитать стоимость всех строк в тиынах."""
    total = 0
    for row in cart["items"]:
        total += row["price"] * row["qty"]
    return total


def discount(cart: dict[str, Any]) -> int:
    """Вычислить лучшую из объёмной скидки и промокода."""
    amount = subtotal(cart)
    volume = 0
    promo = 0
    if amount >= 10000:
        volume = amount * 5 // 100
    if cart["promo"] == "SAVE10":
        promo = amount * 10 // 100
    elif cart["promo"] == "FIX500":
        promo = min(500, amount)
    return max(volume, promo)


def delivery(cart: dict[str, Any]) -> int:
    """Доставка бесплатна от 10000 после скидки, пустая корзина стоит ноль."""
    if not cart["items"]:
        return 0
    net = subtotal(cart) - discount(cart)
    if net >= 10000:
        return 0
    return 700


def quote(cart: dict[str, Any]) -> dict[str, Any]:
    """Сформировать снимок цен и строк корзины."""
    result: dict[str, Any] = {}
    result["items"] = []
    for row in sorted(cart["items"], key=lambda item: item["sku"]):
        copied = {}
        copied["sku"] = row["sku"]
        copied["price"] = row["price"]
        copied["qty"] = row["qty"]
        result["items"].append(copied)
    result["subtotal"] = subtotal(cart)
    result["discount"] = discount(cart)
    result["shipping"] = delivery(cart)
    result["total"] = result["subtotal"] - result["discount"] + result["shipping"]
    result["promo"] = cart["promo"]
    result["closed"] = cart["closed"]
    return result


def checkout(cart: dict[str, Any]) -> None:
    """Оформить непустую корзину."""
    check_open(cart)
    if len(cart["items"]) == 0:
        raise ValueError("empty cart")
    cart["closed"] = True


def process(events: list[dict[str, Any]]) -> dict[str, Any]:
    """Выполнить последовательность действий над изменяемой корзиной."""
    cart = new_cart()
    for event in events:
        operation = event.get("op")
        if operation == "add":
            add_item(cart, event)
        elif operation == "remove":
            remove_item(cart, event)
        elif operation == "promo":
            set_promo(cart, event)
        elif operation == "checkout":
            checkout(cart)
        else:
            raise ValueError("unknown operation")
    return quote(cart)


def run_file(source: Path, target: Path) -> None:
    """Прочитать журнал, выполнить расчёт, записать и напечатать результат."""
    text = source.read_text(encoding="utf-8")
    events = json.loads(text)
    result = process(events)
    report = json.dumps(result, ensure_ascii=False, sort_keys=True)
    target.write_text(report, encoding="utf-8")
    print(report)


src\shop\model.py
from dataclasses import dataclass


@dataclass(frozen=True, order=True)
class Item:
    """Объект-значение товарной строки, деньги в тиынах."""

    sku: str
    price: int
    qty: int


@dataclass(frozen=True)
class Cart:
    """Неизменяемое состояние корзины."""

    items: tuple[Item, ...] = ()
    promo: str = ""
    closed: bool = False


@dataclass(frozen=True)
class Event:
    """Нормализованное действие редьюсера."""

    op: str
    sku: str = ""
    price: int = 0
    qty: int = 1
    code: str = ""


src\shop\pipeline.py
from functools import reduce
from typing import Any

from .core import reduce_event
from .model import Cart, Event
from .pricing import quote


def parse_event(row: dict[str, Any]) -> Event:
    """Нормализовать поля только соответствующей операции."""
    op = str(row.get("op", ""))
    return Event(
        op,
        str(row.get("sku", "")).strip() if op in ("add", "remove") else "",
        int(row.get("price", 0)) if op == "add" else 0,
        int(row.get("qty", 1)) if op in ("add", "remove") else 1,
        str(row.get("code", "")).upper().strip() if op == "promo" else "",
    )


def process(events: list[dict[str, Any]]) -> dict[str, Any]:
    """Лениво разобрать действия и свернуть состояние корзины."""
    return quote(reduce(reduce_event, map(parse_event, events), Cart()))


src\shop\pricing.py
from dataclasses import asdict
from typing import Any

from .discounts import discount, subtotal
from .model import Cart


def quote(cart: Cart) -> dict[str, Any]:
    """Вернуть совместимый отчёт с новыми контейнерами."""
    amount, saved = subtotal(cart), discount(cart)
    shipping = 700 if cart.items and amount - saved < 10000 else 0
    return {
        "items": [asdict(x) for x in cart.items],
        "subtotal": amount,
        "discount": saved,
        "shipping": shipping,
        "total": amount - saved + shipping,
        "promo": cart.promo,
        "closed": cart.closed,
    }

import argparse
import json
from pathlib import Path

from .pipeline import process


def run_file(source: Path, target: Path) -> None:
    """Оставить файловые эффекты и печать в оболочке."""
    result = process(json.loads(source.read_text(encoding="utf-8")))
    report = json.dumps(result, ensure_ascii=False, sort_keys=True)
    target.write_text(report, encoding="utf-8")
    print(report)


def main(argv: list[str] | None = None) -> None:
    """CLI для журнала действий и файла заказа."""
    parser = argparse.ArgumentParser()
    parser.add_argument(
        "source", type=Path, nargs="?", default=Path(__file__).parents[2] / "sample.json"
    )
    parser.add_argument("--output", type=Path, default=Path("order.json"))
    args = parser.parse_args(argv)
    run_file(args.source, args.output)
